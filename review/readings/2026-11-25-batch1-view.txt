  Prop
  :=
  ∀ (b a : ℕ →₀ ℚ) (q : ℚ),
    (b : ℕ → ℚ) 0 = 0 →
      (a : ℕ → ℚ) 0 = 0 →
        PiFormula.value b a = (↑q : ℝ) * Real.pi →
          (∀ (m : ℕ), 2 ≤ m → (b : ℕ → ℚ) m = 0) ∧
            (∀ (m : ℕ), 4 ≤ m → (a : ℕ → ℚ) m = 0) ∧

                q = ((a : ℕ → ℚ) 1 + (a : ℕ → ℚ) 2 - (a : ℕ → ℚ) 3) / 4

def PiFormula.value :
  (ℕ →₀ ℚ) → (ℕ →₀ ℚ) → ℝ
  :=
  fun (b a : ℕ →₀ ℚ) ↦
    (b.sum fun (m : ℕ) (c : ℚ) ↦ (↑c : ℝ) * Real.arctan (1 / 2 ^ m)) +
      a.sum fun (m : ℕ) (c : ℚ) ↦ (↑c : ℝ) * Real.arctan (1 / (2 ^ m - 1))
