-- from module Pilean.PiFormula.LogPrimes
def PiFormula.IsFValue :
  ℝ → Prop
  :=
  fun (x : ℝ) ↦
    ∃ (N : ℕ) (ρ : ℕ → ℚ),
      (0 : ℕ) < N ∧ (∀ (k : ℕ), ρ (k + N) = ρ k) ∧ HasSum (PiFormula.seriesTerm ρ) x

-- from module Pilean.PiFormula.VSeries
def PiFormula.seriesTerm :
  (ℕ → ℚ) → ℕ → ℝ
  :=
  fun (ρ : ℕ → ℚ) (k : ℕ) ↦ (↑(ρ k) : ℝ) / (2 : ℝ) ^ (k / (2 : ℕ)) / (↑k : ℝ)

-- from module Pilean.PiFormula.LogPrimeTest
def PiFormula.primPart :
  ℕ → ℤ
  :=
  fun (ℓ : ℕ) ↦
    if orderOf (2 : ZMod ℓ) % (8 : ℕ) = (4 : ℕ) then
      if (↑ℓ : ℤ) ∣ PiFormula.Aplus (orderOf (2 : ZMod ℓ) / (4 : ℕ)) then
        PiFormula.Aplus (orderOf (2 : ZMod ℓ) / (4 : ℕ))
      else PiFormula.Aminus (orderOf (2 : ZMod ℓ) / (4 : ℕ))
    else Polynomial.eval (2 : ℤ) (Polynomial.cyclotomic (orderOf (2 : ZMod ℓ)) ℤ)

def PiFormula.Aplus :
  ℕ → ℤ
  :=
  fun (n : ℕ) ↦ Zsqrtd.norm (Polynomial.eval PiFormula.tG (Polynomial.cyclotomic n GaussianInt))

def PiFormula.Compendium.BaileyFormula :
  ℝ → ℝ → ℕ → ℕ → (N : ℕ) → (Fin N → ℚ) → Prop
  :=
  fun (x c : ℝ) (b B N : ℕ) (a : Fin N → ℚ) ↦
    ∃ (S : ℝ), PiFormula.Compendium.HasBaileyValue b B N a S ∧ x = c * S

  :=
  fun (b B N : ℕ) (a : Fin N → ℚ) (x : ℝ) ↦ HasSum (PiFormula.BaseB.baileyBlock b B N a) x

def PiFormula.BaseB.baileyBlock :
  ℕ → ℕ → (N : ℕ) → (Fin N → ℚ) → ℕ → ℝ
